# MANIFEST.toml Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an authoritative `MANIFEST.toml` file inventory, a `tools/manifest-query` Perl query tool, and wire both into the Makefile — first as a lint target (Phase 1), then as manifest-driven dist targets (Phase 2).

**Architecture:** TOML file at the repo root using nested-table Approach A (files keyed by path under category sections). A standalone Perl script (`tools/manifest-query`) parses it via TOML::Tiny and answers structured queries. Phase 1 adds `make manifest-lint`; Phase 2 replaces hard-coded file lists in `make dist`, `dist-deploy`, and `dist-minimal`.

**Tech Stack:** Perl 5.20+, TOML::Tiny (CPAN; `libtoml-tiny-perl` / `perl-TOML-Tiny`), Getopt::Long (core), File::Spec (core), Test::More (core), prove

```perl
# run_query: invoke manifest-query with --manifest pointing at $fixture.
# Returns ($exit_code, \@stdout_lines).
sub run_query {
    my (@args) = @_;
    my @cmd = ($^X, $query, '--manifest', $fixture, @args);
    my $out = '';
    open my $fh, '-|', @cmd
        or die "Cannot run manifest-query: $!\n";
    $out .= $_ while <$fh>;
    close $fh;
    return ($? >> 8, [grep { length $_ } split /\n/, $out]);
}

# run_query_dir: like run_query but from a given directory (for --validate).
sub run_query_dir {
    my ($dir, $manifest, @args) = @_;
    my @cmd = ($^X, $query, '--manifest', $manifest, @args);
    my $old = getcwd();
    chdir $dir;
    my $out = '';
    open my $fh, '-|', @cmd
        or do { chdir $old; die "Cannot run manifest-query: $!\n" };
    $out .= $_ while <$fh>;
    close $fh;
    my $exit = $? >> 8;
    chdir $old;
    return ($exit, [grep { length $_ } split /\n/, $out]);
}
```

```perl
#!/usr/bin/perl
# vim: ts=4 sw=4 et sm ai smd sc bg=dark
# perltidy -i=4 -vt=1 -vtc=0 -lp -cti=1 -cab=1 -nbbc -csc -b
########################################################################
# SPDX-License-Identifier: GPL-3.0-or-later
# SPDX-FileCopyrightText: 2026 Christopher Cashell <topher@zyp.org>
########################################################################
# tools/manifest-query - Query MANIFEST.toml for build automation and
#   analysis. Outputs newline-separated file paths or structured text.
#   Requires TOML::Tiny (libtoml-tiny-perl / perl-TOML-Tiny).
#
use strict;
use warnings;
use TOML::Tiny qw(from_toml);
use Getopt::Long qw(:config no_auto_abbrev bundling);
use File::Spec;
use File::Basename qw(dirname);
use Cwd qw(abs_path getcwd);

########################################################################
# Option parsing
########################################################################

my %opt;
GetOptions(\%opt,
    'manifest=s',
    'dist=s',
    'category=s',
    'tag=s',
    'field=s',
    'validate',
    'deps-deb',
    'build-deps-deb',
    'deps-rpm',
    'build-deps-rpm',
    'help',
) or usage(1);
usage(0) if $opt{help};

########################################################################
# Locate and parse MANIFEST.toml
########################################################################

my $manifest_path = _find_manifest($opt{manifest});
die "Cannot find MANIFEST.toml\n" unless defined $manifest_path;

my $toml_text = do {
    open my $fh, '<', $manifest_path
        or die "Cannot open $manifest_path: $!\n";
    local $/;
    <$fh>;
};
my ($manifest, $err) = from_toml($toml_text);
die "Failed to parse $manifest_path: $err\n" if $err;

########################################################################
# Build flat file index: $files{path} = { category => ..., ...fields }
########################################################################

my %files;
for my $cat (sort keys %$manifest) {
    next if $cat eq 'meta';
    my $section = $manifest->{$cat};
    next unless ref $section eq 'HASH';
    for my $path (keys %$section) {
        $files{$path} = { %{$section->{$path}}, category => $cat };
    }
}

########################################################################
# Dispatch
########################################################################

if ($opt{validate}) {
    cmd_validate(\%files);
} elsif ($opt{'deps-deb'}) {
    cmd_deps_deb($manifest->{meta}{deps}, 'runtime');
} elsif ($opt{'build-deps-deb'}) {
    cmd_deps_deb($manifest->{meta}{deps}, 'build');
} elsif ($opt{'deps-rpm'}) {
    cmd_deps_rpm($manifest->{meta}{deps}, 'runtime');
} elsif ($opt{'build-deps-rpm'}) {
    cmd_deps_rpm($manifest->{meta}{deps}, 'build');
} else {
    cmd_list(\%files, \%opt);
}

########################################################################
# Commands
########################################################################

# cmd_validate -- verify every listed file exists relative to cwd
sub cmd_validate {
    my ($files) = @_;
    my @missing = grep { !-e $_ } sort keys %$files;
    if (@missing) {
        warn "Missing files:\n";
        warn "  $_\n" for @missing;
        exit 1;
    }
    printf "OK: %d files validated\n", scalar keys %$files;
    exit 0;
}

# cmd_list -- filter by dist/category/tag, print paths or a named field
sub cmd_list {
    my ($files, $opt) = @_;
    my @results = sort keys %$files;

    if (defined $opt->{dist}) {
        my $want = $opt->{dist};
        @results = grep {
            my $d = $files->{$_}{dist};
            defined $d && ref $d eq 'ARRAY' && grep { $_ eq $want } @$d;
        } @results;
    }

    if (defined $opt->{category}) {
        my $want = $opt->{category};
        @results = grep { ($files->{$_}{category} // '') eq $want } @results;
    }

    if (defined $opt->{tag}) {
        my $want = $opt->{tag};
        @results = grep {
            my $t = $files->{$_}{tags};
            defined $t && ref $t eq 'ARRAY' && grep { $_ eq $want } @$t;
        } @results;
    }

    if (defined $opt->{field}) {
        my $f = $opt->{field};
        for my $path (@results) {
            my $val = $files->{$path}{$f};
            print "$val\n" if defined $val;
        }
    } else {
        print "$_\n" for @results;
    }
    return;
}

# cmd_deps_deb -- emit Debian Depends:/Recommends: or Build-Depends: lines
sub cmd_deps_deb {
    my ($deps, $mode) = @_;
    return unless defined $deps && ref $deps eq 'ARRAY';

    if ($mode eq 'runtime') {
        my (@requires, %or_groups, @recommends);
        for my $dep (@$deps) {
            my $rel = $dep->{relationship} // '';
            my $pkg = $dep->{deb_package}  // next;
            if ($rel eq 'requires') {
                push @requires, defined $dep->{version}
                    ? "$pkg (>= " . _strip_op($dep->{version}) . ")"
                    : $pkg;
            } elsif ($rel eq 'requires_one_of') {
                push @{$or_groups{$dep->{group} // 'misc'}}, $pkg;
            } elsif ($rel eq 'recommends') {
                push @recommends, $pkg;
            }
        }
        for my $g (sort keys %or_groups) {
            push @requires, join(' | ', @{$or_groups{$g}});
        }
        print 'Depends: '    . join(', ', @requires)   . "\n" if @requires;
        print 'Recommends: ' . join(', ', @recommends) . "\n" if @recommends;
    } else {
        my @build = map { $_->{deb_package} }
            grep { ($_->{relationship} // '') =~ /^build_/ && defined $_->{deb_package} }
            @$deps;
        print 'Build-Depends: ' . join(', ', @build) . "\n" if @build;
    }
    return;
}

# cmd_deps_rpm -- emit RPM Requires:/Recommends: or BuildRequires: lines
sub cmd_deps_rpm {
    my ($deps, $mode) = @_;
    return unless defined $deps && ref $deps eq 'ARRAY';

    if ($mode eq 'runtime') {
        for my $dep (@$deps) {
            my $rel = $dep->{relationship} // '';
            my $pkg = $dep->{rpm_package}  // next;
            if ($rel eq 'requires') {
                my $ver = defined $dep->{version}
                    ? " >= " . _strip_op($dep->{version})
                    : '';
                print "Requires: $pkg$ver\n";
            } elsif ($rel eq 'requires_one_of' || $rel eq 'recommends') {
                print "Recommends: $pkg\n";
            }
        }
    } else {
        for my $dep (@$deps) {
            next unless ($dep->{relationship} // '') =~ /^build_/;
            my $pkg = $dep->{rpm_package} // next;
            print "BuildRequires: $pkg\n";
        }
    }
    return;
}

########################################################################
# Helpers
########################################################################

# _find_manifest -- locate MANIFEST.toml: explicit path, or search upward
sub _find_manifest {
    my ($explicit) = @_;
    return $explicit if defined $explicit && -f $explicit;

    my $script_dir = dirname(abs_path($0));
    for my $candidate (
        File::Spec->catfile($script_dir, '..', 'MANIFEST.toml'),
        File::Spec->catfile(getcwd(),    'MANIFEST.toml'),
    ) {
        return File::Spec->canonpath($candidate) if -f $candidate;
    }
    return undef;
}

# _strip_op -- remove leading '>= ' from a version string
sub _strip_op {
    my ($v) = @_;
    $v =~ s{^\s*>=\s*}{};
    return $v;
}

sub usage {
    my ($exit) = @_;
    print <<'END';
Usage: tools/manifest-query [options]

  --manifest PATH     Path to MANIFEST.toml (default: auto-detect)
  --dist DIST         List files in DIST (full|deploy|minimal|deb|rpm|arch)
  --category CAT      Filter by category
  --tag TAG           Filter by tag
  --field FIELD       Output named field value instead of file path
  --validate          Check all listed files exist (relative to cwd)
  --deps-deb          Emit Debian Depends:/Recommends: lines
  --build-deps-deb    Emit Debian Build-Depends: lines
  --deps-rpm          Emit RPM Requires:/Recommends: lines
  --build-deps-rpm    Emit RPM BuildRequires: lines
  --help              Show this message
END
    exit $exit;
}
```
